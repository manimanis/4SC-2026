# 1. Fonction de primalité sans break
def est_premier(n):
    if n < 2:
        return False
    i = 2
    premier = True
    while i * i <= n and premier:
        if n % i == 0:
            premier = False
        i = i + 1
    return premier

# 2. Fonction PGCD (Algorithme d'Euclide)
def pgcd(a, b):
    while b != 0:
        r = a % b
        a = b
        b = r
    return a

# 3. Fonction somme des chiffres
def somme_chiffres(n):
    s = 0
    temp = n
    while temp > 0:
        s = s + (temp % 10)
        temp = temp // 10
    return s

# --- Programme Principal ---
a = 0
while not (a > 0):
    a = int(input("Donner un entier A positif : "))

b = 0
while not (b > 0):
    b = int(input("Donner un entier B positif : "))

print("A est-il premier ? :", est_premier(a))
print("PGCD(A, B) =", pgcd(a, b))
print("Somme des chiffres de A =", somme_chiffres(a))
Conformité Officielle
  • Portée locale : Les variables i, premier, r, s, temp sont strictement locales à chaque fonction.
  • Retour de type simple : Chaque fonction renvoie une seule valeur scalaire (bool ou int).
  • Pas de break : Le contrôle d'arrêt du while respecte la condition logique sans rupture de flux.
