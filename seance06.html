from numpy import array

# 1. Module de saisie de la taille N (sans break)
def saisie_taille(borne_min, borne_max):
    taille = 0
    while not (borne_min <= taille <= borne_max):
        taille = int(input(f"Donner la taille du tableau ({borne_min} à {borne_max}) : "))
    return taille

# 2. Module de remplissage (t mutable passé par adresse)
def remplir_tab(t, n):
    for i in range(n):
        t[i] = int(input(f"Donner l'élément t[{i}] : "))

# 3. Module d'affichage (élément par élément)
def afficher_tab(t, n):
    print("Contenu du tableau :")
    for i in range(n):
        print(f"t[{i}] = {t[i]}")

# 4. Fonction de calcul de la moyenne
def calculer_moyenne(t, n):
    somme = 0
    for i in range(n):
        somme = somme + t[i]
    return somme / n

# --- Programme Principal ---
# Déclaration statique numpy de 10 entiers
t = array([0] * 10)

n = saisie_taille(3, 10)
remplir_tab(t, n)
afficher_tab(t, n)

moy = calculer_moyenne(t, n)
print(f"Moyenne des éléments = {moy:.2f}")
Conformité Officielle
  • T.D.N.T. obligatoire : Le type Tab = Tableau de 10 Entier est déclaré dans le T.D.N.T., puis utilisé dans les en-têtes (@t: Tab).
  • Bibliothèque numpy : Déclaration obligatoire via from numpy import array et t = array([0] * 10).
  • Affichage indicé : Interdiction formelle du print(t) direct ; affichage boucle élément par élément.
  • Passage par adresse : En algorithmique via @t: Tab, en Python transmis naturellement par référence d'objet.
